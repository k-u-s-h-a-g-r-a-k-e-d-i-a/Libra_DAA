cmake_minimum_required(VERSION 3.16)
project(LibraDAA_C LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(WIN32)
  add_compile_definitions(__USE_MINGW_ANSI_STDIO=1)
else()
  add_compile_definitions(_POSIX_C_SOURCE=200809L)
endif()
add_library(daa STATIC src/algorithms.c src/workbench.c)
target_include_directories(daa PUBLIC src)
add_executable(libradaa src/main.c src/library.c src/evidence.c src/tests.c)
target_link_libraries(libradaa PRIVATE daa)
if(NOT MSVC)
  target_link_libraries(libradaa PRIVATE m)
endif()
enable_testing()
add_test(NAME regression COMMAND libradaa --test)
set_tests_properties(regression PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
